# Linux/arm64-safe dependency subset for the Murder Bot worker container.
#
# IMPORTANT: the repo's top-level requirements.txt pins mlx==0.32.0 and
# mlx-vlm==0.6.6. Those are Apple-Silicon (Metal) only and CANNOT install on
# Linux/arm64. They are intentionally omitted here, which means the local
# vision-LLM fallback (`run_bot.py --llm-fallback`) is unavailable on the VM.
# The core CV + OCR automation loop does not need them. If you want an LLM
# fallback on the server, swap it for a remote API call — do not add mlx here.
#
# opencv-python is swapped for the -headless build (no X/GL libs, server-correct).
# Pinned for the bot image's Python 3.11 base (numpy 2.5.x needs 3.12; opencv 5.x
# lacks py3.11 wheels). These are the known-good Linux versions.
numpy==2.1.3
pillow==11.0.0
opencv-python-headless==4.10.0.84
rapidocr-onnxruntime==1.2.3
onnxruntime==1.19.2
pytesseract
psycopg2-binary==2.9.12
